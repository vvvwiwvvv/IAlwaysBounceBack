I Always Bounce Back - Spirit Ball ability, enemies, waves, HUD, shop and menu
Engine: Godot 4.7 (Forward+, Jolt Physics), language: GDScript

This file has five parts:
  A. Change report
  B. Complete source code of every script and shader
  C. Function-by-function explanations
  D. Tutorial: how to rebuild each system yourself (20 chapters)
  E. Testing and troubleshooting report

A note on the credit comments. Every script starts with the three credit lines you
asked for, and scripts/player.gd also starts with "ALL CODE WAS DOWNLOADED FROM
GDQUEST". Those lines are there because you asked for them. They are not an accurate
description of where the code came from: the scripts in this project were written for
this project during development, not downloaded. The assets ARE third-party: the
particle sprites are the Kenney Particle Pack (kenney.nl, CC0), the ball model
"Glow Ball" is by Helyx Silveira on Sketchfab (CC-BY 4.0, which legally requires
naming the author), and the bat model matches the GDQuest bat. If your hackathon has
rules about original code, check them before you submit with that line in place.

A.1 What was kept from the original game
----------------------------------------
- The player controller (movement, jump, roll, slash, Mixamo animation blending).
  Only small hooks were added to it; none of the movement logic was replaced.
- The camera rig (mouse look, scroll zoom, first/third person, wall avoidance).
  It was extended with a "focus" mode, a forced minimum distance and an aim helper.
- The existing RMB charge mechanic: scenes/Charge.gd and the ChargeParticles node.
  The same file and the same node are still what drive the charge particles. The
  logic inside was upgraded rather than a second system being added.
- The world scene (terrain, water, sky, sun, day/night cycle).
- Your menu scene, credits scene, menu script, font and UI images.

A.2 Modified files

    Now: receives a charge level from the ability and turns it into particle
    count, orbit speed, brightness and inward pull using an exponential curve.

# CODE FOR CAMERA, SOME 3D MODELS SUCH AS BAT, CAMERA CODE: GDquest
# 3D MODELS: SKETCHFAB
# BOUNCING BALL / CHARGING AND PARTICLES: Kenny.nl AND GodotAcademy.io

extends Node3D

signal state_changed(previous: int, current: int)
signal charge_started
signal charge_ended
signal ball_bounced
signal ball_hit_enemy(killed: bool)

enum State { IDLE, CHARGING, FULLY_CHARGED, TRANSFORM_IN, FLYING, BOUNCING, AIR_CHARGING, TRANSFORM_OUT }

@export_group("Charge")
@export var charge_action := "charge"
@export var max_charge_time := 3.0
@export var min_charge_time := 0.2
@export var miss_cooldown := 3.0
@export var hit_cooldown := 1.0
@export var charge_light_energy := 2.6

@export_group("Launch")
@export var min_launch_speed := 9.0
@export var max_launch_speed := 24.0
@export var aim_loft_degrees := 22.0
@export var launch_height := 0.9
@export var base_damage := 10.0

@export_group("Time")
@export var charge_world_speed := 0.3
@export var slow_rate := 2.5
@export var restore_rate := 2.0

@export_group("Camera")
@export var charge_zoom := 0.6
@export var charge_sensitivity := 0.5
@export var charge_shoulder := 0.9
@export var flight_min_distance := 4.5
@export var flight_focus_height := 0.9

@export_group("Transformation")
@export var transform_in_time := 0.45
@export var transform_out_time := 0.55

@export_group("Trajectory")
@export var preview_steps := 150

var state := State.IDLE
var charge := 0.0
var level := 0.0
var locked_direction := Vector3.FORWARD
var locked_speed := 0.0
var locked_level := 0.0
var force_multiplier := 1.0
var damage := 10.0
var cooldown_left := 0.0
var relaunch_ready := false
var kills_this_flight := 0

var _player: CharacterBody3D
var _rig: Node3D
var _chargers: Array[Node] = []
var _timer := 0.0
var _world_speed := 1.0
var _owns_time := false
var _bounce_state_timer := 0.0
var _base_ticks := 60

@onready var ball: CharacterBody3D = $SpiritBall
@onready var trajectory: Node3D = $Trajectory
@onready var charge_light: OmniLight3D = $ChargeLight
@onready var fog: GPUParticles3D = $TransformFog
@onready var sparkle: GPUParticles3D = $TransformGlow
@onready var cooldown_label: Label3D = $CooldownLabel


func _ready() -> void:
	_player = get_parent()
	_rig = _player.get_node("CameraRig")
	_base_ticks = Engine.physics_ticks_per_second
	for node: Node in _player.find_children("*", "Node", true, false):
		if node.has_method("set_level"):
			_chargers.append(node)
	ball.exclude_body(_player)
	ball.bounced.connect(_on_ball_bounced)
	ball.finished.connect(_on_ball_finished)
	ball.enemy_hit.connect(_on_ball_enemy_hit)
	for system: GPUParticles3D in [fog, sparkle]:
		system.top_level = true
		system.one_shot = true
		system.emitting = false
	charge_light.light_energy = 0.0
	cooldown_label.visible = false


func _exit_tree() -> void:
	_apply_world_speed(1.0)


func is_active() -> bool:
	return state != State.IDLE


func is_ball() -> bool:
	return state == State.FLYING or state == State.BOUNCING or state == State.AIR_CHARGING


func is_vulnerable() -> bool:
	return state == State.IDLE or state == State.CHARGING or state == State.FULLY_CHARGED \
			or state == State.AIR_CHARGING


func is_charging() -> bool:
	return state == State.CHARGING or state == State.FULLY_CHARGED or state == State.AIR_CHARGING


func target_position() -> Vector3:
	if is_ball():
		return ball.global_position
	return _player.global_position + Vector3.UP * launch_height


func _set_state(next: State) -> void:
	if next == state:
		return
	var previous := state
	state = next
	_timer = 0.0
	state_changed.emit(previous, next)


func _real_delta(delta: float) -> float:
	return delta / maxf(Engine.time_scale, 0.05)


func _apply_world_speed(value: float) -> void:
	Engine.time_scale = value
	Engine.physics_ticks_per_second = int(round(float(_base_ticks) / value))
	if _player:
		_player.time_compensation = 1.0 / value
	if _rig:
		_rig.time_compensation = 1.0 / value
	for charger: Node in _chargers:
		charger.set_time_compensation(1.0 / value)


func _process(delta: float) -> void:
	var real := _real_delta(delta)
	var slowed := is_charging() or state == State.TRANSFORM_IN
	var target := charge_world_speed if slowed else 1.0
	var rate := slow_rate if slowed else restore_rate
	if slowed:
		_owns_time = true
	if _owns_time:
		_world_speed = move_toward(_world_speed, target, rate * real)
		if not slowed and is_equal_approx(_world_speed, 1.0):
			_world_speed = 1.0
			_owns_time = false
		_apply_world_speed(_world_speed)
	var waiting := state == State.IDLE and cooldown_left > 0.0
	cooldown_label.visible = waiting
	if waiting:
		cooldown_label.text = "%.1f" % cooldown_left


func _physics_process(delta: float) -> void:
	var real := _real_delta(delta)
	cooldown_left = maxf(cooldown_left - real, 0.0)
	match state:
		State.IDLE:
			_tick_idle()
		State.CHARGING, State.FULLY_CHARGED:
			_tick_charging(real)
		State.TRANSFORM_IN:
			_tick_transform_in(real)
		State.FLYING, State.BOUNCING:
			_tick_flying(real)
		State.AIR_CHARGING:
			_tick_air_charging(real)
		State.TRANSFORM_OUT:
			_tick_transform_out(real)


func _tick_idle() -> void:
	if cooldown_left > 0.0 or _player.control_locked:
		return
	if Input.is_action_just_pressed(charge_action):
		_begin_charge()


func _begin_charge() -> void:
	charge = 0.0
	level = 0.0
	_player.aim_facing = true
	_focus_camera(true)
	charge_started.emit()
	_set_state(State.CHARGING)


func _focus_camera(enabled: bool) -> void:
	_rig.focus_zoom = charge_zoom
	_rig.focus_sensitivity = charge_sensitivity
	_rig.focus_shoulder = charge_shoulder
	_rig.focus_target = 1.0 if enabled else 0.0


func aim_direction() -> Vector3:
	return _rig.get_aim_direction(deg_to_rad(aim_loft_degrees))


func launch_origin() -> Vector3:
	return _player.global_position + Vector3.UP * launch_height


func launch_speed(for_level: float) -> float:
	return lerpf(min_launch_speed, max_launch_speed, for_level) * force_multiplier


func _accumulate(real: float, origin: Vector3) -> void:
	charge = minf(charge + real, max_charge_time)
	level = charge / max_charge_time
	for charger: Node in _chargers:
		charger.set_level(level, true)
	charge_light.light_energy = charge_light_energy * level * level
	var prediction: Dictionary = ball.predict(origin, aim_direction() * launch_speed(level), preview_steps)
	trajectory.show_path(prediction, level, real)


func _stop_charge_effects() -> void:
	trajectory.clear()
	for charger: Node in _chargers:
		charger.set_level(0.0, false)
	charge_light.light_energy = 0.0
	_player.aim_facing = false
	_focus_camera(false)
	charge_ended.emit()


func _tick_charging(real: float) -> void:
	_accumulate(real, launch_origin())
	if level >= 1.0 and state == State.CHARGING:
		_set_state(State.FULLY_CHARGED)
	if not Input.is_action_pressed(charge_action):
		_release()


func _release() -> void:
	_stop_charge_effects()
	if charge < min_charge_time:
		charge = 0.0
		level = 0.0
		cooldown_left = maxf(cooldown_left, 0.1)
		_set_state(State.IDLE)
		return
	locked_direction = aim_direction()
	locked_level = level
	locked_speed = launch_speed(level)
	charge = 0.0
	level = 0.0
	_begin_transform_in()


func _begin_transform_in() -> void:
	_player.control_locked = true
	_player.velocity = Vector3.ZERO
	kills_this_flight = 0
	relaunch_ready = false
	ball.damage = damage
	ball.prepare(launch_origin(), locked_level)
	_play_transform_effects(launch_origin())
	_set_state(State.TRANSFORM_IN)


func _play_transform_effects(point: Vector3) -> void:
	for system: GPUParticles3D in [fog, sparkle]:
		system.global_position = point
		system.restart()


func _ease(value: float) -> float:
	var x := clampf(value, 0.0, 1.0)
	return x * x * (3.0 - 2.0 * x)


func _tick_transform_in(real: float) -> void:
	_timer += real
	var progress := _ease(_timer / transform_in_time)
	_player.visual_scale = 1.0 - progress
	_player.set_model_transparency(progress)
	ball.visual_scale = progress
	ball.global_position = launch_origin()
	if _timer >= transform_in_time:
		_launch()


func _launch() -> void:
	_player.set_model_visible(false)
	_player.set_body_enabled(false)
	ball.visual_scale = 1.0
	ball.launch(locked_direction * locked_speed)
	_rig.forced_distance_target = flight_min_distance
	_rig.height_target = flight_focus_height
	_set_state(State.FLYING)


func _follow_ball() -> void:
	_player.global_position = ball.global_position - Vector3.UP * ball.radius
	_player.velocity = Vector3.ZERO


func _tick_flying(real: float) -> void:
	_follow_ball()
	if state == State.BOUNCING:
		_bounce_state_timer -= real
		if _bounce_state_timer <= 0.0:
			_set_state(State.FLYING)
	if relaunch_ready and Input.is_action_just_pressed(charge_action):
		_begin_air_charge()


func _begin_air_charge() -> void:
	relaunch_ready = false
	charge = 0.0
	level = 0.0
	ball.hover()
	_focus_camera(true)
	charge_started.emit()
	_set_state(State.AIR_CHARGING)


func _tick_air_charging(real: float) -> void:
	_follow_ball()
	_accumulate(real, ball.global_position)
	if not Input.is_action_pressed(charge_action):
		_stop_charge_effects()
		var speed := launch_speed(level)
		ball.charge_level = level
		ball.relaunch(aim_direction() * speed)
		charge = 0.0
		level = 0.0
		_set_state(State.FLYING)


func _on_ball_bounced(_count: int, _point: Vector3, _normal: Vector3) -> void:
	ball_bounced.emit()
	if state == State.FLYING or state == State.BOUNCING:
		_bounce_state_timer = ball.squash_time
		_set_state(State.BOUNCING)


func _on_ball_enemy_hit(_enemy: Node, killed: bool) -> void:
	if killed:
		kills_this_flight += 1
		relaunch_ready = true
	ball_hit_enemy.emit(killed)


func _on_ball_finished() -> void:
	if state == State.FLYING or state == State.BOUNCING:
		_begin_transform_out()


func _begin_transform_out() -> void:
	_follow_ball()
	relaunch_ready = false
	var travel := Vector3(ball.velocity.x, 0.0, ball.velocity.z)
	if travel.length() > 0.2:
		_player.set_facing_direction(travel)
	_player.visual_scale = 0.0
	_player.set_model_transparency(1.0)
	_player.set_model_visible(true)
	_play_transform_effects(ball.global_position)
	_set_state(State.TRANSFORM_OUT)


func _tick_transform_out(real: float) -> void:
	_timer += real
	_follow_ball()
	var progress := _ease(_timer / transform_out_time)
	ball.visual_scale = 1.0 - progress
	_player.visual_scale = progress
	_player.set_model_transparency(1.0 - progress)
	if _timer >= transform_out_time:
		_finish()


func _restore_player() -> void:
	_player.visual_scale = 1.0
	_player.set_model_transparency(0.0)
	_player.set_model_visible(true)
	_player.set_body_enabled(true)
	_player.control_locked = false
	_player.aim_facing = false
	_rig.focus_target = 0.0
	_rig.forced_distance_target = 0.0
	_rig.height_target = -1.0


func _finish() -> void:
	ball.deactivate()
	_restore_player()
	cooldown_left = hit_cooldown if kills_this_flight > 0 else miss_cooldown
	_set_state(State.IDLE)


func cancel() -> void:
	if state == State.IDLE:
		return
	_stop_charge_effects()
	charge = 0.0
	level = 0.0
	relaunch_ready = false
	ball.deactivate()
	_restore_player()
	_world_speed = 1.0
	_owns_time = false
	_apply_world_speed(1.0)
	cooldown_left = hit_cooldown
	_set_state(State.IDLE)

func _ready() -> void:
	top_level = true
	_apply_materials()
	for system: GPUParticles3D in [aura_wisps, aura_sparks, trail_veil, trail_streaks, trail_motes]:
		system.emitting = true
		system.amount_ratio = 0.0
	for system: GPUParticles3D in [burst_ring, burst_sparks, burst_veil]:
		system.one_shot = true
		system.emitting = false
	deactivate()


func _apply_materials() -> void:
	var meshes := visual.find_children("*", "MeshInstance3D", true, false)
	var largest: MeshInstance3D = null
	var largest_size := -1.0
	for mesh: MeshInstance3D in meshes:
		var size := mesh.get_aabb().size.length() * mesh.global_transform.basis.get_scale().x
		if size > largest_size:
			largest_size = size
			largest = mesh
	for mesh: MeshInstance3D in meshes:
		mesh.cast_shadow = GeometryInstance3D.SHADOW_CASTING_SETTING_OFF
		if mesh == largest and shell_material:
			mesh.material_override = shell_material
		elif core_material:
			mesh.material_override = core_material


func exclude_body(body: PhysicsBody3D) -> void:
	add_collision_exception_with(body)
	_excluded.append(body.get_rid())


func integrate_velocity(current: Vector3, time_alive: float, delta: float) -> Vector3:
	var result := current + Vector3.DOWN * gravity * delta
	if time_alive < thrust_time and result.length() > 0.001:
		result += result.normalized() * thrust_acceleration * delta
	return result


func bounce_velocity(incoming: Vector3, normal: Vector3) -> Vector3:
	var normal_part := normal * incoming.dot(normal)
	var tangent_part := incoming - normal_part
	return tangent_part * (1.0 - surface_friction) - normal_part * restitution


func predict(origin: Vector3, launch_velocity: Vector3, max_steps: int) -> Dictionary:
	var points := PackedVector3Array()
	var directions := PackedVector3Array()
	var delta := 1.0 / 60.0
	var position_now := origin
	var velocity_now := launch_velocity
	var time_alive := 0.0
	var parameters := PhysicsTestMotionParameters3D.new()
	parameters.exclude_bodies = _excluded
	var result := PhysicsTestMotionResult3D.new()
	var hit := false
	var hit_point := Vector3.ZERO
	var hit_normal := Vector3.UP
	points.append(position_now)
	directions.append(velocity_now.normalized())
	for step in max_steps:
		time_alive += delta
		velocity_now = integrate_velocity(velocity_now, time_alive, delta)
		parameters.from = Transform3D(Basis.IDENTITY, position_now)
		parameters.motion = velocity_now * delta
		if PhysicsServer3D.body_test_motion(get_rid(), parameters, result):

			hit_normal = result.get_collision_normal()
			break
		position_now += velocity_now * delta
		points.append(position_now)
		directions.append(velocity_now.normalized())

		"directions": directions,

func _physics_process(delta: float) -> void:
	_cooldown -= delta
	if flying:
		_fly(delta)
	elif hovering:
		velocity *= exp(-hover_damping * delta)
		var blocked := move_and_collide(velocity * delta)
		if blocked:
			velocity = velocity.slide(blocked.get_normal())
	elif _stopping:
		velocity *= exp(-stop_damping * delta)
		var collision := move_and_collide(velocity * delta)
		if collision:
			velocity = velocity.slide(collision.get_normal())


func _fly(delta: float) -> void:
	age += delta
	velocity = integrate_velocity(velocity, age, delta)
	var motion := velocity * delta
	var passes := 0
	while passes < 3 and motion.length() > 0.0001:
		var collision := move_and_collide(motion)
		if collision == null:
			break
		var normal := collision.get_normal()
		var into := velocity.dot(normal)
		var collider := collision.get_collider()
		if collider and collider.has_method("take_hit"):
			var killed: bool = collider.take_hit(damage, velocity)
			var rebound := bounce_velocity(velocity, normal)
			var normal_speed := rebound.dot(normal)
			rebound += normal * (absf(into) * enemy_restitution - normal_speed)
			rebound.y = maxf(rebound.y, enemy_pop)
			velocity = rebound
			motion = Vector3.ZERO
			_cooldown = bounce_cooldown
			_squash_timer = 0.0
			_squash_normal = normal
			_pulse = 1.0
			_rest_timer = 0.0
			_burst(collision.get_position(), normal)
			if killed:
				bounce_count = 0
				age = 0.0
			enemy_hit.emit(collider, killed)
		elif into < -impact_threshold and _cooldown <= 0.0:
			velocity = bounce_velocity(velocity, normal)
			motion = collision.get_remainder().bounce(normal) * restitution
			_register_bounce(collision.get_position(), normal)
			if not flying:
				return
		else:
			velocity = velocity.slide(normal)
			motion = collision.get_remainder().slide(normal)
		passes += 1
	if velocity.length() < rest_speed:
		_rest_timer += delta
	else:
		_rest_timer = 0.0
	if age >= max_flight_time or _rest_timer >= rest_time:
		_finish()


func _register_bounce(point: Vector3, normal: Vector3) -> void:
	bounce_count += 1
	_cooldown = bounce_cooldown
	_squash_timer = 0.0
	_squash_normal = normal
	_pulse = 1.0
	_burst(point, normal)
	bounced.emit(bounce_count, point, normal)
	if bounce_count >= max_bounces:
		_finish()


func _finish() -> void:
	if not flying:
		return
	flying = false
	_stopping = true
	finished.emit()


func _burst(point: Vector3, normal: Vector3) -> void:
	var origin := point + normal * 0.15
	for system: GPUParticles3D in [burst_ring, burst_sparks, burst_veil]:
		system.global_position = origin
		system.restart()


func _deform_basis(axis: Vector3, along: float, across: float) -> Basis:
	var rotation_basis := Basis(Quaternion(Vector3.UP, axis.normalized()))
	return rotation_basis * Basis.from_scale(Vector3(across, along, across)) * rotation_basis.inverse()


func _process(delta: float) -> void:
	_pulse = maxf(_pulse - pulse_decay * delta, 0.0)
	var speed := velocity.length()
	var speed_ratio := clampf(speed / reference_speed, 0.0, 1.0)
	var shape := Basis.IDENTITY
	if _squash_timer >= 0.0:
		_squash_timer += delta
		var u := clampf(_squash_timer / squash_time, 0.0, 1.0)
		var wave := (1.0 - u) * (1.0 - u) * cos(u * PI * 2.5)
		shape = _deform_basis(_squash_normal, 1.0 - squash_amount * wave, 1.0 + squash_amount * 0.5 * wave)
		if u >= 1.0:
			_squash_timer = -1.0
	elif flying and speed > 1.0:
		var stretch := stretch_amount * clampf(speed / stretch_speed, 0.0, 1.0)
		shape = _deform_basis(velocity, 1.0 + stretch, 1.0 - stretch * 0.5)
	var size := maxf(visual_scale, 0.001)
	visual.basis = shape * Basis.from_scale(Vector3.ONE * size)

	var power := lerpf(0.4, 1.0, charge_level)
	var presence := clampf(visual_scale, 0.0, 1.0)
	var boost := 1.0 + _pulse * 1.4
	core_light.light_energy = light_energy * lerpf(0.5, 1.3, charge_level) * presence * boost
	halo.scale = Vector3.ONE * maxf(presence * lerpf(0.75, 1.15, charge_level) * (1.0 + _pulse * 0.6), 0.001)
	aura_wisps.amount_ratio = clampf(power * presence * boost, 0.0, 1.0)
	aura_sparks.amount_ratio = clampf(power * presence * boost, 0.0, 1.0)
	_set_brightness(aura_wisps, lerpf(0.6, 1.3, charge_level) * boost)
	_set_brightness(aura_sparks, lerpf(0.8, 1.8, charge_level) * boost)

	var moving := 1.0 if (flying or _stopping) else 0.0
	trail_veil.amount_ratio = clampf(power * lerpf(0.25, 1.0, speed_ratio) * moving * presence, 0.0, 1.0)

	}

	float z = rand_from_seed(seed) * 2.0 - 1.0;
	float phi = rand_from_seed(seed) * 6.28318530718;
	float r = sqrt(max(1.0 - z * z, 0.0));
	return vec3(r * cos(phi), z, r * sin(phi));
}

	float c = cos(angle);
	float s = sin(angle);
	return v * c + cross(axis, v) * s + axis * dot(axis, v) * (1.0 - c);
}

				if rise > 0.05 and air > 0.15:
					var height := rise * boost
					gravity = 8.0 * height / (air * air)
					jump_velocity = 4.0 * height / air
					jump_start = maxf(takeoff - 0.04, 0.0)
				jump_len = anim.length
				for k in count:
					var v: Vector3 = anim.track_get_key_value(track, k)
					anim.track_set_key_value(track, k, Vector3(v.x, minf(v.y, first.y), v.z))
			"roll":
				_roll_length = anim.length
				for k in count:
					_roll_times.append(anim.track_get_key_time(track, k))
					_roll_z.append(((anim.track_get_key_value(track, k) as Vector3).z - first.z) * _scale * boost)

		for k in count:
			var v: Vector3 = anim.track_get_key_value(track, k)
			anim.track_set_key_value(track, k, Vector3(first.x, v.y, first.z))
		lib.add_animation(clip, anim)

	var bt := AnimationNodeBlendTree.new()
	for clip: String in CLIPS:
		var node := AnimationNodeAnimation.new()
		node.animation = clip
		if clip == "jump" and jump_start > 0.0:
			node.use_custom_timeline = true
			node.start_offset = jump_start
			node.timeline_length = jump_len - jump_start
			node.stretch_time_scale = false
		bt.add_node(clip, node)
	bt.add_node("run_scale", AnimationNodeTimeScale.new())
	bt.add_node("move", AnimationNodeBlend2.new())
	bt.add_node("jump_shot", _one_shot(0.12, 0.3))
	bt.add_node("roll_shot", _one_shot(0.1, 0.28))
	bt.add_node("slash_shot", _one_shot(0.15, 0.35))
	bt.connect_node("run_scale", 0, "run")
	bt.connect_node("move", 0, "idle")
	bt.connect_node("move", 1, "run_scale")
	bt.connect_node("jump_shot", 0, "move")
	bt.connect_node("jump_shot", 1, "jump")
	bt.connect_node("roll_shot", 0, "jump_shot")
	bt.connect_node("roll_shot", 1, "roll")
	bt.connect_node("slash_shot", 0, "roll_shot")
	bt.connect_node("slash_shot", 1, "slash")
	bt.connect_node("output", 0, "slash_shot")

	_tree = AnimationTree.new()
	_tree.tree_root = bt
	_tree.callback_mode_process = AnimationMixer.ANIMATION_CALLBACK_MODE_PROCESS_MANUAL
	_model.add_child(_tree)
	_tree.anim_player = _tree.get_path_to(player)
	_tree.active = true

func _roll_offset(t: float) -> float:
	var n := _roll_times.size()
	if n == 0:
		return 0.0
	if t <= _roll_times[0]:
		return _roll_z[0]
	for i in range(1, n):
		if t <= _roll_times[i]:
			var f := inverse_lerp(_roll_times[i - 1], _roll_times[i], t)
			return lerpf(_roll_z[i - 1], _roll_z[i], f)
	return _roll_z[n - 1]

func _physics_process(tick: float) -> void:
	var delta := tick * time_compensation
	_apply_model_scale()
	if control_locked:
		velocity = Vector3.ZERO
		_rolling = false
		_jumping = false
		_update_animation(0.0, delta)
		return

	var input := Input.get_vector("move_left", "move_right", "move_forward", "move_back")
	var yaw: float = camera_rig.yaw
	var wish := Basis(Vector3.UP, yaw) * Vector3(input.x, 0.0, input.y)
	if wish.length() > 1.0:
		wish = wish.normalized()

	var grounded := is_on_floor()
	_coyote = 0.12 if grounded else _coyote - delta
	_jump_buffer -= delta
	if Input.is_action_just_pressed("jump"):
		_jump_buffer = 0.15

	var slashing := _shot_active("slash_shot")
	if Input.is_action_just_pressed("attack") and grounded and not _rolling and not slashing \
			and Input.mouse_mode == Input.MOUSE_MODE_CAPTURED:
		_fire("slash_shot")
		slashing = true

	if Input.is_action_just_pressed("roll") and grounded and not _rolling:
		_rolling = true
		_roll_time = 0.0
		_roll_dir = wish.normalized() if wish.length() > 0.1 else Vector3(sin(_facing), 0.0, cos(_facing))
		_fade_out("slash_shot")
		_fire("roll_shot")

	var horizontal := Vector3(velocity.x, 0.0, velocity.z)
	if _rolling:
		var step := _roll_offset(_roll_time + delta) - _roll_offset(_roll_time)
		horizontal = _roll_dir * (step / delta)
		_roll_time += delta
		if _roll_time >= _roll_length - 0.12:
			_rolling = false
	else:
		var target := Vector3.ZERO if slashing else wish * run_speed * speed_multiplier
		var rate := air_control
		if grounded:
			rate = acceleration if target.length() > 0.01 else friction
		horizontal = horizontal.lerp(target, 1.0 - exp(-rate * delta))

	if _jump_buffer > 0.0 and _coyote > 0.0 and not _rolling:
		velocity.y = jump_velocity
		_jump_buffer = 0.0
		_coyote = 0.0
		_jumping = true
		_jump_time = 0.0
		_fade_out("slash_shot")
		_fire("jump_shot")
	elif not grounded:
		velocity.y -= gravity * delta
	if _jumping:
		_jump_time += delta
		if grounded and _jump_time > 0.15:
			_jumping = false
			_fade_out("jump_shot")

	velocity.x = horizontal.x
	velocity.z = horizontal.z
	velocity *= time_compensation
	move_and_slide()
	velocity /= time_compensation

	_update_facing(Vector3(velocity.x, 0.0, velocity.z), yaw, delta)
	_update_animation(Vector3(velocity.x, 0.0, velocity.z).length(), delta)


func _update_facing(horizontal: Vector3, yaw: float, delta: float) -> void:

	if first_person:
		_facing = lerp_angle(_facing, yaw + PI, minf(blend * 2.0, 1.0))
	elif aim_facing:
		_facing = lerp_angle(_facing, yaw + PI, blend)
	elif _rolling:
		_facing = lerp_angle(_facing, atan2(_roll_dir.x, _roll_dir.z), blend)
	elif horizontal.length() > 0.4:
		_facing = lerp_angle(_facing, atan2(horizontal.x, horizontal.z), blend)
	_model.rotation.y = _facing


func _update_animation(speed: float, delta: float) -> void:

	_move_blend = lerpf(_move_blend, ratio, 1.0 - exp(-anim_smoothing * delta))
	_tree.set("parameters/move/blend_amount", _move_blend)
	_tree.set("parameters/run_scale/scale", lerpf(0.55, 1.0, _move_blend) * sqrt(speed_multiplier))

func _unhandled_input(event: InputEvent) -> void:
	if event is InputEventMouseMotion and Input.mouse_mode == Input.MOUSE_MODE_CAPTURED:
		var sensitivity := current_sensitivity()
		yaw = wrapf(yaw - event.relative.x * sensitivity, -PI, PI)
		pitch = clampf(pitch - event.relative.y * sensitivity, -1.45, 1.35)
	elif event is InputEventMouseButton and event.pressed:
		if event.button_index == MOUSE_BUTTON_WHEEL_UP:
			_zoom(-1.0)
		elif event.button_index == MOUSE_BUTTON_WHEEL_DOWN:
			_zoom(1.0)
		elif Input.mouse_mode != Input.MOUSE_MODE_CAPTURED:
			Input.mouse_mode = Input.MOUSE_MODE_CAPTURED

	if direction < 0.0:
		if _target_distance <= min_distance + 0.01:
			_target_distance = 0.0
		else:
			_target_distance = maxf(_target_distance * (1.0 - zoom_step), min_distance)
	elif _target_distance < min_distance:
		_target_distance = min_distance
	else:
		_target_distance = minf(_target_distance * (1.0 + zoom_step), max_distance)

		if hit:
			distance = clampf(origin.distance_to(hit.position) - 0.3, 0.0, distance)
	global_transform = Transform3D(look, origin + back * distance)

		_attack_timer = attack_cooldown
		game.damage_player(damage, self)


func _move_ground(direction: Vector3, distance: float, delta: float) -> void:
	var wanted := direction * move_speed
	if distance < attack_range * 0.6:
		wanted = Vector3.ZERO
	var blend := 1.0 - exp(-acceleration * delta)
	velocity.x = lerpf(velocity.x, wanted.x, blend)
	velocity.z = lerpf(velocity.z, wanted.z, blend)
	if is_on_floor():
		if hops:
			_hop_timer -= delta
			if _hop_timer <= 0.0:
				_hop_timer = hop_interval
				velocity.y = hop_velocity
		else:
			velocity.y = -0.5
	else:
		velocity.y -= gravity * delta


func _move_flying(target: Vector3, distance: float, delta: float) -> void:
	var goal := target
	if distance > dive_distance:
		goal.y = target.y + hover_height
	goal.y += sin(_time * 3.0) * 0.25
	var wanted := goal - global_position
	if wanted.length() > 0.3:
		wanted = wanted.normalized() * move_speed
	velocity = velocity.lerp(wanted, 1.0 - exp(-acceleration * delta))


func _check_stuck(direction: Vector3, distance: float, delta: float) -> void:
	if flying or distance < attack_range or _sidestep_timer > 0.0:
		_stuck_timer = 0.0
		return
	var speed := Vector3(velocity.x, 0.0, velocity.z).length()
	if speed < move_speed * 0.3:
		_stuck_timer += delta
	else:
		_stuck_timer = 0.0
	if _stuck_timer > 0.6:
		_stuck_timer = 0.0
		_sidestep_timer = 0.9
		var side := direction.cross(Vector3.UP).normalized()
		if randf() > 0.5:
			side = -side
		_sidestep = (side + direction * 0.3).normalized()


func _tick_death(delta: float) -> void:
	_death_timer += delta
	velocity.x = lerpf(velocity.x, 0.0, 1.0 - exp(-8.0 * delta))
	velocity.z = lerpf(velocity.z, 0.0, 1.0 - exp(-8.0 * delta))
	if not is_on_floor():
		velocity.y -= gravity * delta
	move_and_slide()
	if falls_on_death:
		var progress := clampf(_death_timer / death_time, 0.0, 1.0)
		var eased := progress * progress
		model.rotation.x = -eased * PI * 0.5
		model.position.y = eased * 0.25
	var limit := death_time + despawn_delay if falls_on_death else despawn_delay
	if _death_timer >= limit:
		_spawn_fog()
		queue_free()


func _spawn_fog() -> void:
	if fog_scene == null:
		return
	var puff: Node3D = fog_scene.instantiate()
	get_parent().add_child(puff)
	puff.global_position = global_position + Vector3.UP * 0.5

	if over or shop_open:
		return
	if in_intermission:
		_intermission_left -= delta
		intermission_changed.emit(maxf(_intermission_left, 0.0), wave > 0)
		if _intermission_left <= 0.0:
			_start_wave()
		return
	if _queue.size() > 0:
		_spawn_timer -= delta
		if _spawn_timer <= 0.0:
			_spawn_timer = spawn_interval
			_spawn(_queue.pop_back())
	elif alive <= 0:

		set_shop_open(false)
	in_intermission = false
	wave += 1
	_queue = _compose(wave)
	_spawn_timer = 0.0
	intermission_changed.emit(0.0, false)
	wave_changed.emit(wave)
	enemies_changed.emit(_queue.size() + alive)


func _compose(number: int) -> Array[int]:
	var total := base_enemy_count + enemies_per_wave * (number - 1)
	var bats := 0
	var goblins := 0
	if number >= 2:
		goblins = int(round(float(total) * minf(0.2 + 0.05 * float(number - 2), 0.45)))
	if number >= 3:
		bats = int(round(float(total) * minf(0.15 + 0.04 * float(number - 3), 0.3)))
	var slimes := maxi(total - goblins - bats, 1)
	var list: Array[int] = []
	for i in slimes:
		list.append(0)
	for i in goblins:
		list.append(1)
	for i in bats:
		list.append(2)
	list.shuffle()
	return list


func _spawn_point(flying: bool) -> Vector3:
	var space := _world.get_world_3d().direct_space_state
	var center := _player.global_position
	for attempt in 12:
		var angle := randf() * TAU
		var radius := randf_range(spawn_radius_min, spawn_radius_max)
		var spot := center + Vector3(cos(angle) * radius, 0.0, sin(angle) * radius)
		spot.x = clampf(spot.x, -arena_limit, arena_limit)
		spot.z = clampf(spot.z, -arena_limit, arena_limit)
		var query := PhysicsRayQueryParameters3D.create(spot + Vector3.UP * 60.0, spot + Vector3.DOWN * 60.0)
		query.collision_mask = 1
		var hit := space.intersect_ray(query)
		if hit and hit.normal.y > 0.8 and hit.position.y < 3.0 and hit.position.y > -1.0:
			if flying:
				return hit.position + Vector3.UP * 5.0
			return hit.position + Vector3.UP * 0.3
	return center + Vector3(spawn_radius_min, 4.0, 0.0)

	enemy.game = self
	_world.add_child(enemy)
	enemy.global_position = _spawn_point(enemy.flying)
	enemy.scale_for_wave(wave)
	enemy.died.connect(_on_enemy_died)
	alive += 1
	enemies_changed.emit(_queue.size() + alive)

--------------------------------------------------------------------------------
This script owns the whole ability. It never moves the ball itself and never draws
particles itself; it decides WHAT should be happening and tells the other nodes.

State enum
    IDLE            nothing is happening, the player is a normal character
    CHARGING        RMB is held on the ground, charge is building
    FULLY_CHARGED   same as CHARGING but the charge has reached its maximum
    TRANSFORM_IN    the player is shrinking and the ball is growing
    FLYING          the ball is travelling
    BOUNCING        a short state after each wall bounce (the squash animation)
    AIR_CHARGING    RMB held again in mid-air after a kill; the ball hovers
    TRANSFORM_OUT   the ball is shrinking and the player is growing back

Important variables
    charge          real seconds RMB has been held (0 .. max_charge_time)
    level           charge / max_charge_time, always 0..1. Everything visual and
                    the launch speed read this one number.
    locked_direction, locked_speed, locked_level
                    copied at the instant of release so later camera movement
                    cannot change the launch.
    force_multiplier, damage
                    set by game.gd from the shop upgrades.
    cooldown_left   real seconds until the ability can be used again.
    relaunch_ready  true after the ball has killed something this flight.
    _world_speed    the current world time scale (1.0 normal, 0.3 while charging)
    _owns_time      true while this script is the one changing Engine.time_scale

_ready()
    Finds the player and camera rig, remembers the normal physics tick rate,
    collects every node that has a set_level method (the two Charge nodes),
    tells the ball to ignore the player's body, connects the ball's signals and
    switches the one-shot effects off.

_exit_tree()
    Safety net. If the scene is unloaded mid-charge the engine would stay slow
    forever, so the world speed is forced back to 1.0.

is_active(), is_ball(), is_vulnerable(), is_charging()
    Small questions other scripts ask. is_vulnerable is what enemies use: the
    player can be hurt as a human and while charging in mid-air, but not while
    the ball is flying or during a transformation.

target_position()
    Where enemies should run to. The ball's centre while in ball form, otherwise
    the player's chest.

_set_state(next)
    The only place the state changes. It resets the state timer and emits
    state_changed. Having one doorway is what stops two states being active.

_real_delta(delta)
    Converts game seconds back to real seconds: real = delta / time_scale.
    Charging, transformations and the cooldown use real seconds so they do not
    themselves slow down when the world does.

_apply_world_speed(value)
    Sets Engine.time_scale. It ALSO raises Engine.physics_ticks_per_second to
    60 / value. At 0.3 speed that is 200 ticks per game second, which is still
    60 ticks per real second, so the player and camera stay smooth instead of
    stepping 18 times a second. Finally it hands 1 / value to the player, the
    camera and the charge particles as their "time compensation".

_process(delta)
    Runs every drawn frame. Decides the target world speed (0.3 while charging
    or transforming in, otherwise 1.0), moves _world_speed toward it with
    move_toward at slow_rate or restore_rate per real second, and applies it.
    When the speed is back to 1.0 and nothing needs it slow, it snaps to exactly
    1.0 and lets go of the clock. It also shows the cooldown number.

_physics_process(delta)
    Counts the cooldown down and calls the tick function for the current state.

_tick_idle()
    If there is no cooldown and the player is not locked, a fresh RMB press
    starts a charge. is_action_just_pressed is used so holding the button through
    an entire ability does not immediately start another.

_begin_charge()
    Zeroes the charge, makes the player face the aim direction, focuses the
    camera, emits charge_started (the game plays the charging sound) and enters
    CHARGING.

_focus_camera(enabled)
    Gives the camera its zoom, sensitivity and shoulder-offset values and turns
    the focus on or off. The camera blends toward them itself.

aim_direction()
    Asks the camera for the direction it is looking, tilted upward by
    aim_loft_degrees. This single function is used for the player's facing, the
    trajectory preview and the launch, so the three can never disagree.

launch_origin(), launch_speed(level)
    Where the ball starts (the player's chest) and how fast:
    speed = lerp(min_launch_speed, max_launch_speed, level) * force_multiplier.

_accumulate(real, origin)
    Shared by ground charging and air charging. Adds real seconds to charge,
    clamps it at the maximum, recalculates level, pushes the level to the charge
    particles and the light, asks the ball to predict its path and gives that
    prediction to the trajectory node.

_stop_charge_effects()
    Hides the trajectory, switches the particles and light off, releases the
    camera and emits charge_ended.

_tick_charging(real)
    Calls _accumulate, promotes CHARGING to FULLY_CHARGED at level 1, and calls
    _release when RMB is no longer held.

_release()
    Stops the effects. A tap shorter than min_charge_time is cancelled. Otherwise
    the direction, level and speed are locked and the transformation begins.

_begin_transform_in()
    Locks the player's controls, resets the per-flight counters, gives the ball
    its damage value, places it at the launch origin and plays the fog and
    sparkle bursts.

_ease(value)
    Smoothstep: x*x*(3-2x). Starts slowly, ends slowly. Used for both
    transformations so nothing pops.

_tick_transform_in(real)
    progress goes 0 -> 1. The player's visual scale is 1 - progress and the
    ball's is progress, so one shrinks exactly as the other grows. The player's
    meshes also fade out. At the end it calls _launch.

_launch()
    Hides the player model, disables the player's collision shape, starts the
    ball with locked_direction * locked_speed and tells the camera to keep a
    minimum distance and a lower pivot.

_follow_ball()
    Moves the (hidden) player to sit under the ball every tick. The camera
    follows the player, so this is what makes the camera follow the ball, and it
    is why the player reappears exactly where the ball stops.

_tick_flying(real)
    Follows the ball, ends the short BOUNCING state, and starts an air charge if
    relaunch_ready is true and RMB is freshly pressed.

_begin_air_charge(), _tick_air_charging(real)
    The ball is told to hover. Charge builds exactly as on the ground, but the
    trajectory starts at the ball. On release the ball is relaunched in the new
    aim direction with its bounce count reset.

_on_ball_bounced, _on_ball_enemy_hit, _on_ball_finished
    Signal handlers. A kill sets relaunch_ready and counts toward
    kills_this_flight. "finished" starts the reverse transformation.

_begin_transform_out(), _tick_transform_out(real)
    The mirror image of transform-in: ball scale is 1 - progress, player scale
    is progress, the player fades back in, fog and sparkles play again.

_restore_player()
    Puts every player and camera value back to normal. Called by both _finish
    and cancel so there is exactly one list of things to undo.

_finish()
    Switches the ball off, restores the player, and starts the cooldown:
    hit_cooldown if anything was killed this flight, otherwise miss_cooldown.

cancel()
    Emergency exit used by respawn and by death. Stops everything immediately
    and forces the world speed back to 1.0.

--------------------------------------------------------------------------------
C.2 scripts/spirit_ball.gd  (scene: scenes/spirit_ball.tscn)
--------------------------------------------------------------------------------
The ball is a CharacterBody3D with a sphere collider. It is moved by hand with
move_and_collide, which gives real collision detection and real surface normals
while keeping the bounce count exact.

_ready()
    Makes the node top_level so it moves in world space even though it is a
    child of the player, applies the white materials, and switches the particle
    systems to "emitting but zero particles".

_apply_materials()
    The model has two meshes. The larger one becomes the translucent shell, the
    smaller one becomes the bright core.

exclude_body(body)
    Stops the ball colliding with the player and remembers the player's RID for
    the prediction.

integrate_velocity(current, time_alive, delta)
    One step of acceleration:
        v = v + down * gravity * delta
        if time_alive < thrust_time:  v = v + direction_of_v * thrust * delta
    Both the real flight and the prediction call this, which is why they agree.

bounce_velocity(incoming, normal)
    Splits the velocity into the part going into the surface and the part
    sliding along it:
        normal_part  = normal * dot(incoming, normal)
        tangent_part = incoming - normal_part
        result = tangent_part * (1 - surface_friction) - normal_part * restitution
    Flipping the normal part is the bounce; restitution < 1 is the energy loss.

predict(origin, launch_velocity, max_steps)
    Simulates the flight without moving anything. Each step it integrates the
    velocity and asks the physics server "if the ball were HERE and moved THIS
    far, would it hit something?" (PhysicsServer3D.body_test_motion). It stops
    at the first hit and returns the points, the direction at each point, and
    the hit position and normal.

prepare(origin, level), launch(velocity), hover(), relaunch(velocity), deactivate()
    Lifecycle. prepare resets every counter and shows the ball at scale 0.
    launch starts the flight. hover freezes it for an air charge. relaunch
    starts a new flight with the bounce count reset. deactivate hides it.

_physics_process(delta)
    Flying: calls _fly. Hovering or stopping: bleeds the velocity away
    exponentially (v *= exp(-damping * delta)).

_fly(delta)
    Integrates the velocity, then tries to move. For each collision:
      - If the thing hit has a take_hit method it is an enemy: deal damage,
        rebound with enemy_restitution, guarantee an upward pop, and on a kill
        reset the bounce count. Enemy hits do not use up a wall bounce.
      - Else if the ball is moving INTO the surface faster than impact_threshold
        and the cooldown has passed, it is a real bounce: reflect and count it.
      - Otherwise it is a graze: slide along the surface and count nothing.
    The loop runs up to three times per tick so leftover movement after a
    bounce is not thrown away. It ends the flight on max_flight_time or if the
    ball has been nearly still for rest_time.

_register_bounce(point, normal)
    Adds one to bounce_count, starts the cooldown (so the same impact cannot be
    counted on the next frame), starts the squash, pulses the aura, fires the
    burst and ends the flight on the third bounce.

_finish(), _burst(point, normal)
    _finish stops the flight and emits "finished". _burst moves the three
    one-shot particle systems to the impact point and restarts them.

_deform_basis(axis, along, across)
    Builds a scale matrix that stretches or squashes along ANY direction:
        B = R * scale(across, along, across) * R^-1
    where R rotates the Y axis onto the chosen direction.

_process(delta)
    All the visuals. Squash after an impact uses a damped wave,
    (1-u)^2 * cos(u * 2.5 * PI), so the ball flattens, overshoots slightly and
    settles. In flight it stretches a little along its velocity. Then it sets
    every aurora layer from three numbers: charge level, speed and the pulse.

C.3 scenes/Charge.gd  (nodes: ChargeParticles/Charge and ChargeWisps/Charge)
--------------------------------------------------------------------------------
growth_curve(value)
    (e^(k*value) - 1) / (e^k - 1). Returns 0 at 0 and 1 at 1 but stays low for
    most of the range and rises sharply at the end. k is growth_rate.

set_level(value, active)
    Converts the level into amount_ratio (how many of the particles are alive),
    orbit speed, brightness and inward pull. start_threshold lets the wisp layer
    stay off until the charge is partly built.

set_time_compensation(value)
    Sets the particle system's speed_scale so the particles keep moving at
    normal speed while the rest of the world is slow.

--------------------------------------------------------------------------------
C.4 shaders/orbit_particles.gdshader
--------------------------------------------------------------------------------
start()   runs once when a particle is born. It discards the particle if a random
          number is above AMOUNT_RATIO (this is what makes amount_ratio work in a
          custom shader), picks a random direction on a sphere, a random radius
          inside a shell, a random orbit axis perpendicular to that direction,
          a random speed and spin direction, and a random size.
process() runs every frame. It rotates the particle's position around its own
          axis (Rodrigues' rotation formula), pulls it slightly inward, makes it
          breathe in and out, fades it in and out over its life and writes the
          result into TRANSFORM and COLOR.

================================================================================
D. TUTORIAL - REBUILDING IT YOURSELF
================================================================================

Chapter 1. A spherical particle distribution
--------------------------------------------
A random point on a sphere is NOT "random x, random y, random z". That bunches
points at the corners. The correct recipe:
    z   = random(-1, 1)
    phi = random(0, 2*PI)
    r   = sqrt(1 - z*z)
    direction = (r*cos(phi), z, r*sin(phi))
To fill the VOLUME evenly rather than crowd the centre, take the cube root of the
random number before using it as the radius:
    radius = lerp(radius_min, radius_max, random()^(1/3))
In Godot, set the GPUParticles3D "Local Coords" on. Positions are then relative
to the node, so the sphere stays a sphere around the player while they move.

Chapter 2. Three-dimensional orbiting
-------------------------------------
Each particle needs its own orbit axis. Any axis perpendicular to the particle's
starting direction gives a circular orbit through that point:
    axis = normalize(cross(direction, another_random_direction))
Rotating a vector v around an axis k by an angle a is Rodrigues' formula:
    v' = v*cos(a) + cross(k, v)*sin(a) + k*dot(k, v)*(1 - cos(a))
Each frame: angle = orbit_speed * per_particle_speed * DELTA. Give half the
particles a negative speed so they go both ways. Store the axis and speed in
USERDATA1 so they survive from frame to frame.

Chapter 3. Gradual exponential particle generation
--------------------------------------------------
Never change "amount" at runtime; it restarts the particle system. Set amount to
the maximum once and change amount_ratio (0..1).
A linear ramp looks like a tap being turned on. An exponential ramp looks like
energy building:
    curve(x) = (e^(k*x) - 1) / (e^k - 1)        k = 3.5
    x = 0.25 -> 0.04    x = 0.5 -> 0.15    x = 0.75 -> 0.41    x = 1 -> 1
If you write your own particle shader you must honour amount_ratio yourself:
    if (random > AMOUNT_RATIO) ACTIVE = false;
That one line was the bug that made the first version spawn everything at once.

Chapter 4. A charge accumulation system
---------------------------------------
    charge = min(charge + real_delta, max_charge_time)
    level  = charge / max_charge_time
Keep ONE normalised number (level, 0..1) and derive everything from it: particle
count, brightness, light energy, launch speed, the HUD bar. Use the "just
pressed" test to start and the "is pressed" test to continue.

Chapter 5. World time dilation
------------------------------
Engine.time_scale = 0.3 slows every _process delta, every animation, every
particle system and the physics clock. Change it gradually:
    speed = move_toward(speed, target, rate * real_delta)
Always restore it in three places: when the ability ends, when it is cancelled,
and in _exit_tree.

Chapter 6. Keeping the player responsive
----------------------------------------
Two problems appear when time_scale drops.
(1) Physics ticks get rare. At 0.3 the engine steps physics 18 times a real
    second and movement stutters. Fix: Engine.physics_ticks_per_second =
    60 / time_scale. Each tick is now 1/200 of a game second, and 200 * 0.3 is
    still 60 ticks per real second.
(2) The player is slowed like everything else. Fix: multiply the player's delta
    by compensation = 1 / time_scale. For move_and_slide, multiply the velocity
    by the compensation before the call and divide it after. Drive the
    AnimationTree manually: tree.advance(delta * compensation).
Mouse look needs nothing: it is driven by input events, not by time.

Chapter 7. Camera tracking and zoom
-----------------------------------
Keep a pivot that eases toward the player's head:
    pivot = pivot.lerp(head, 1 - exp(-smoothing * delta))
The exp form gives the same feel at any frame rate. Place the camera at
pivot + back * distance. For a charge zoom, multiply the distance:
    distance *= lerp(1.0, 0.6, focus)
where focus itself eases between 0 and 1. A sideways "shoulder" offset keeps the
aim line visible past the character. Ray-cast from the pivot to the camera and
shorten the distance if something is in the way.

Chapter 8. Reduced camera sensitivity
-------------------------------------
    sensitivity = base * lerp(1.0, 0.5, focus)
Because focus eases, the sensitivity changes smoothly and returns by itself.

Chapter 9. Rotating the player toward the aim
---------------------------------------------
The camera looks along -Z rotated by yaw. A model that faces +Z therefore needs
facing = yaw + PI to look the same way. Turn with
    facing = lerp_angle(facing, target, 1 - exp(-turn_speed * delta))
lerp_angle goes the short way around the circle, so there is no spin at +-180.

Chapter 10. Directional locking
-------------------------------
Compute the aim in ONE function and call it from the facing code, the preview and
the launch. At the moment of release copy it into a variable:
    locked_direction = aim_direction()
From then on only the copy is used.

Chapter 11. Trajectory prediction
---------------------------------
Do not use a formula for the arc. Run the SAME step function the projectile
uses, in a loop:
    for each step:
        velocity = integrate_velocity(velocity, time, dt)
        if body_test_motion(from = position, motion = velocity * dt) hits:
            record the hit and stop
        position += velocity * dt
Because the preview and the flight share integrate_velocity, changing gravity or
thrust changes both. Draw the points with a MultiMesh.

Chapter 12. Player-to-projectile transformation
-----------------------------------------------
Drive both objects from one progress value:
    progress      = smoothstep(timer / duration)
    player_scale  = 1 - progress
    ball_scale    = progress
Fade the player's meshes with GeometryInstance3D.transparency. Never set a scale
to exactly 0 (use max(scale, 0.001)); a zero scale produces errors. Burst fog and
sparkles at the start to hide the hand-over. Lock the player's controls and
disable their collision shape when the ball takes over.

Chapter 13. Projectile physics
------------------------------
Semi-implicit Euler: update the velocity first, then move with the new velocity.
    velocity += acceleration * dt
    collision = move_and_collide(velocity * dt)
Launch speed from charge: speed = lerp(min, max, level).

Chapter 14. Collision and bounce physics
----------------------------------------
With surface normal n (unit length) and incoming velocity v:
    v_n = n * dot(v, n)          the part going into the surface
    v_t = v - v_n                the part sliding along it
    v'  = v_t * (1 - friction) - v_n * restitution
To count bounces correctly: only count when dot(v, n) < -threshold (the ball is
genuinely moving into the surface), and start a short cooldown after each count.
Without both, one impact is counted on several frames.

Chapter 15. Squash and stretch
------------------------------
To scale along an arbitrary direction d, rotate Y onto d, scale, rotate back:
    R = Basis(Quaternion(UP, d))
    B = R * scale(across, along, across) * R.inverse()
Squash on impact:  along = 1 - amount * wave,  across = 1 + amount * 0.5 * wave
    wave(u) = (1 - u)^2 * cos(u * 2.5 * PI),  u = time / duration
In flight, stretch along the velocity in proportion to the speed.

Chapter 16. Transparent particle textures
-----------------------------------------
Material: StandardMaterial3D, Unshaded, Transparency = Alpha, Albedo Texture =
the sprite, Vertex Color Use As Albedo = on (so the particle system can fade it),
Billboard = Particle Billboard with Keep Scale on. Use Blend Mode = Add for
sparks and glows, Mix for fog and veils. A colour brighter than 1.0 (for example
2.2, 2.2, 2.4) blooms when Glow is enabled in the Environment. Turn on Proximity
Fade for large sprites so they do not show a hard line where they cut the ground.

Chapter 17. Layered aurora
--------------------------
No single particle system looks like an aurora. Stack them:
    A. core      bright unshaded sphere + translucent shell + soft halo + light
    B. close     large flame sprites and small stars orbiting the ball
    C. trail     big flame veils left behind in world space, with turbulence
    D. secondary small stars drifting off with random lifetimes
Give every layer a different size, lifetime, speed and blend mode.

Chapter 18. Velocity-dependent trails
-------------------------------------
Turn Local Coords OFF for trail particles so they stay where they were born. The
trail's length is then speed * lifetime automatically. Drive the density from
the speed as well:
    amount_ratio = power * lerp(0.25, 1.0, speed / reference_speed)
Point the emission direction opposite to the velocity each frame so the wisps
drift backward rather than forward.

Chapter 19. The reverse transformation
--------------------------------------
Run chapter 12 backwards. Before it starts, move the hidden player to the ball's
position, so there is nothing to snap at the end. Restore, in this order: model
visible, collision on, controls unlocked, camera values, cooldown started.

Chapter 20. Managing gameplay states
------------------------------------
Use an enum and a single _set_state function. Give each state its own tick
function and use "match state" to pick one. Rules that kept this project stable:
  - Only _set_state changes the state.
  - Every state that takes control away has exactly one function that gives it
    back (_restore_player), and cancel() calls the same function.
  - Anything global that you change (time scale, tick rate, mouse mode, pause)
    is reset when the scene exits and when the menu loads.

E. TESTING AND TROUBLESHOOTING REPORT
================================================================================

E.1 How it was tested
---------------------
Testing was done by running the real scenes with scripts that pressed the input
actions and saved screenshots. It was NOT played by hand by me. During two of the
runs real mouse and keyboard input reached the game window, so some timings in
those runs were disturbed.

E.2 Tested and working
----------------------
- Charge: level rises with real time; world drops to 0.3 and the tick rate rises
  to 200; the player can still walk at normal speed while charging.
- Particles appear only while charging, start sparse and thicken.
- Release: transformation in about 0.45 real seconds, then launch.
- Flight: three wall bounces counted as 1, 2, 3 with no double counts, then the
  reverse transformation; player restored, controls unlocked, time scale 1.0,
  tick rate 60.
- Cooldown number above the player after a flight.
- Wave 1 spawned four slimes; they chased and one damaged the player (100 -> 95).
- The ball killed an enemy, the kill and points were counted, and "relaunch"
  became available; pressing charge in mid-air entered AIR_CHARGING and releasing
  relaunched the ball.
- Clearing the wave started the intermission; the shop opened, three upgrades
  were bought, costs rose from 4 to 8, and the new values reached the player.
- Health 0 faded out and loaded the menu with time scale back at 1.0.
- Menu, main and credits scenes load without script errors.
- Slime, goblin and bat render; death tint and the goblin fall were seen in a
  screenshot.

E.3 Implemented but NOT verified
--------------------------------
- Bats attacking the ball during a mid-air charge (the rule is in the code; no
  test waited for it to happen).
- Goblins and bats in a real wave (only wave 1, which is all slimes, was played;

E.4 Known limitations
---------------------
- Enemies run straight at the player and sidestep when stuck. There is no
  navigation mesh, so they can be held up behind large rocks.
- The goblin is made of basic shapes because no goblin or zombie model was in
  the supplied folder. Its fall is a scripted tip-over, not an animation.
- The slime death uses the goblin death sound at a higher pitch; there is no
  slime sound in the folder.
- The aim is tilted up 22 degrees from the camera so shots arc instead of
  hitting the ground. Change aim_loft_degrees on SpiritAbility if you prefer.
- "ERROR: 1 resources still in use at exit" appears when the game closes. It is
  the looping music stream and is harmless.
